(function () {
  /* ============================================================
     PRODUCTION DATA — update these by hand as the work progresses.
     Each stage is a percent 0–100. Stages are weighted equally.
     ============================================================ */
  var STAGES = ['Script', 'Boards', 'Layout', 'Animation', 'Comp'];
  var EPISODES = [
    { id: 'pilot', name: 'Pilot — "Fell From the Sky"', tag: 'Origin', pct: { Script: 55, Boards: 20, Layout: 0, Animation: 0, Comp: 0 } },
    { id: 'e1', name: 'Ch.1 — Checker Town',            tag: 'S1',     pct: { Script: 40, Boards: 0,  Layout: 0, Animation: 0, Comp: 0 } },
    { id: 'e2', name: 'Ch.2 — First Move',              tag: 'S1',     pct: { Script: 15, Boards: 0,  Layout: 0, Animation: 0, Comp: 0 } },
    { id: 'e3', name: 'Ch.3 — The Journey',             tag: 'S1',     pct: { Script: 5,  Boards: 0,  Layout: 0, Animation: 0, Comp: 0 } }
  ];

  /* ============================================================
     FRAME-THE-SCENE polls — swap in real shot mockups any time.
     ============================================================ */
  var POLLS = [
    { id: 'open-shot', q: 'The opening shot of the pilot — how do we meet Checker Town?',
      a: { label: 'Wide & quiet', art: '🌌  a vast night sky, one streak of light falling', note: 'Mystery first — the crash is a rumor.' },
      b: { label: 'Close & loud', art: '💥  smash-cut to the wreck, pieces everywhere', note: 'Action first — drop us in the chaos.' } },
    { id: 'princess-key', q: "Princess's color key:",
      a: { label: 'Warm violet', art: '🟣  regal purples, gold trim', note: 'Storybook royalty.' },
      b: { label: 'Cool slate', art: '🔵  cooler blues, silver trim', note: 'Sci-fi castaway.' } }
  ];

  function esc(s){ return String(s).replace(/[&<>"]/g, function(c){ return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]; }); }
  function epPct(ep){ var t = 0; STAGES.forEach(function(s){ t += (ep.pct[s] || 0); }); return Math.round(t / STAGES.length); }

  // ---- tracker ----
  function renderTracker() {
    var wrap = document.getElementById('pr-eps'); wrap.innerHTML = '';
    var sum = 0;
    EPISODES.forEach(function (ep) {
      sum += epPct(ep);
      var stageRows = STAGES.map(function (s) {
        var v = ep.pct[s] || 0;
        return '<div class="pr-stage"><span class="pr-stage-lab">' + s + '</span>' +
          '<span class="pr-stage-bar"><i style="width:' + v + '%"></i></span>' +
          '<span class="pr-stage-pct">' + v + '%</span></div>';
      }).join('');
      var el = document.createElement('div'); el.className = 'pr-ep';
      el.innerHTML =
        '<div class="pr-ep-head"><div><span class="pr-ep-tag">' + esc(ep.tag) + '</span> ' +
        '<b class="pr-ep-name">' + esc(ep.name) + '</b></div>' +
        '<div class="pr-ep-pct">' + epPct(ep) + '%</div></div>' + stageRows;
      wrap.appendChild(el);
    });
    var overall = EPISODES.length ? Math.round(sum / EPISODES.length) : 0;
    document.getElementById('pr-overall-num').textContent = overall + '%';
    document.getElementById('pr-overall-fill').style.width = overall + '%';
  }

  // ---- frame the scene (local vote) ----
  function voteKey(id){ return 'pjcc.frame.' + id; }
  function getVote(id){ try { return localStorage.getItem(voteKey(id)); } catch (e) { return null; } }
  function setVote(id, side){ try { localStorage.setItem(voteKey(id), side); } catch (e) {} }
  // illustrative baseline tallies so the bars aren't empty before community voting exists
  var SEED = { 'open-shot': { a: 61, b: 47 }, 'princess-key': { a: 73, b: 38 } };

  function renderPolls() {
    var wrap = document.getElementById('pr-poll'); wrap.innerHTML = '';
    POLLS.forEach(function (p) {
      var mine = getVote(p.id);
      var base = SEED[p.id] || { a: 0, b: 0 };
      var av = base.a + (mine === 'a' ? 1 : 0), bv = base.b + (mine === 'b' ? 1 : 0), tot = av + bv || 1;
      function opt(side, o, v) {
        var pctv = Math.round(v / tot * 100);
        return '<button class="pr-opt' + (mine === side ? ' chosen' : '') + '" data-poll="' + p.id + '" data-side="' + side + '">' +
          '<div class="pr-opt-art">' + esc(o.art) + '</div>' +
          '<div class="pr-opt-label">' + esc(o.label) + '</div>' +
          '<div class="pr-opt-note">' + esc(o.note) + '</div>' +
          '<div class="pr-opt-bar' + (mine ? ' show' : '') + '"><i style="width:' + pctv + '%"></i><span>' + pctv + '%</span></div>' +
          '</button>';
      }
      var el = document.createElement('div'); el.className = 'pr-poll-card';
      el.innerHTML = '<div class="pr-poll-q">' + esc(p.q) + (mine ? ' <span class="pr-voted">✓ voted</span>' : '') + '</div>' +
        '<div class="pr-opts">' + opt('a', p.a, av) + opt('b', p.b, bv) + '</div>';
      wrap.appendChild(el);
    });
    Array.prototype.forEach.call(wrap.querySelectorAll('.pr-opt'), function (btn) {
      btn.onclick = function () { setVote(btn.dataset.poll, btn.dataset.side); renderPolls(); };
    });
  }

  renderTracker();
  renderPolls();
})();
</script>

(function () {
  /* ============================================================
     PILOT ANIMATIC — the storyboard reel. Each shot carries a
     composition, an action line, a scratch-VO subtitle, and a
     temp-score motif (played live via PJCCLeitmotif). Swap the
     emoji "art" for real board scans whenever they exist.
     ============================================================ */
  var SHOTS = [
    { id:'s1', slug:'SC 01 · SH 01', comp:'WIDE', art:'🌌', bg:'linear-gradient(180deg,#0a1230,#1b2a5e)',
      action:'Night over the open sea. The screen holds black, then one streak of light falls.',
      vo:{ who:'NARRATOR', line:'They say the sky used to give things. Pieces. Light. Once — a passenger.' },
      motif:'Narrator', dur:5.5 },
    { id:'s2', slug:'SC 01 · SH 02', comp:'WIDE', art:'🛸', bg:'linear-gradient(180deg,#100a26,#3a1d4e)',
      action:'A construction vessel on final approach over Checker Town. Engines cut. Something is jettisoned.',
      vo:null, motif:'Interplanetary Construction Co.', dur:4.5 },
    { id:'s3', slug:'SC 01 · SH 03', comp:'CLOSE', art:'📦', bg:'linear-gradient(180deg,#161033,#2d1b69)',
      action:'A small pod tumbles free, end over end, down toward the dark below.',
      vo:{ who:'NARRATOR', line:'One passenger was never on the manifest.' },
      motif:'Princess', dur:4.5 },
    { id:'s4', slug:'SC 02 · SH 01', comp:'WIDE', art:'🌅', bg:'linear-gradient(180deg,#2a1a5e,#e0995a)',
      action:'Dawn over Checker Town. Checker pieces lie scattered in the dust like fallen debris.',
      vo:{ who:'FATHER', line:'Another quiet morning in Checker Town. And then it wasn’t.' },
      motif:'Father', dur:5.5 },
    { id:'s5', slug:'SC 02 · SH 02', comp:'OTS', art:'🐾', bg:'linear-gradient(180deg,#3a2a1a,#c98a4a)',
      action:'Over the shoulder — a young man kneels in the checker-dust and eases the pod open.',
      vo:{ who:'FATHER', line:'Hello there. …Where on earth did you come from?' },
      motif:'Father', dur:5 },
    { id:'s6', slug:'SC 02 · SH 03', comp:'CLOSE', art:'🐶', bg:'linear-gradient(180deg,#241a3c,#5b2d8e)',
      action:'Close on the pup. Her eyes open. First light. Her first look at him.',
      vo:null, motif:'Princess', dur:4 },
    { id:'s7', slug:'SC 03 · SH 01', comp:'WIDE', art:'♞', bg:'linear-gradient(180deg,#0a0a22,#2a2050)',
      action:'Months later. Princess noses a knight across a battered board — and it lands on the right square.',
      vo:{ who:'NARRATOR', line:'She learned the knight first. Nobody taught her.' },
      motif:'Princess', dur:5 },
    { id:'s8', slug:'SC 03 · SH 02', comp:'WIDE', art:'🏙️', bg:'linear-gradient(180deg,#0a0a22,#1a8fe3)',
      action:'Princess looks out past the water toward the gleam of Chess City. End on the horizon.',
      vo:{ who:'PRINCESS', line:'There. That’s where we’re going.' },
      motif:'Princess', dur:5.5, end:true }
  ];

  // Pinned director's notes — always shown above community comments.
  var PINNED = {
    s1: [{ name:'Director', text:'Hold the black as long as we dare before the streak. Silence sells it.' }],
    s4: [{ name:'Director', text:'The checker pieces in the dust = the show’s thesis in one frame. Don’t rush past it.' }],
    s8: [{ name:'Director', text:'This is the poster shot. If this frame doesn’t give chills, the pilot isn’t done.' }]
  };

  function $(id){ return document.getElementById(id); }
  function esc(s){ return String(s).replace(/[&<>"]/g, function(c){ return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]; }); }

  var frame=$('an-frame'), art=$('an-art'), slug=$('an-slug'), comp=$('an-comp'),
      vo=$('an-vo'), score=$('an-score'), fill=$('an-progress-fill'), count=$('an-count'),
      strip=$('an-strip'), playBtn=$('an-play'), muteBtn=$('an-mute');
  if (!frame) return;

  var cur=0, playing=false, shotT=0, raf=null, muted=false, lastTs=0;

  function motif(name){ if (!muted && window.PJCCLeitmotif) { try { PJCCLeitmotif.play(name); } catch(e){} } }

  function showShot(i, playMotif){
    cur = ((i % SHOTS.length) + SHOTS.length) % SHOTS.length;
    var s = SHOTS[cur];
    frame.style.background = s.bg;
    art.textContent = s.art;
    void art.offsetWidth; art.style.animation='none'; void art.offsetWidth; art.style.animation=''; // retrigger pop
    slug.textContent = s.slug;
    comp.textContent = s.comp;
    frame.classList.toggle('is-end', !!s.end);
    if (s.vo){ vo.innerHTML = '<span class="an-vo-who">'+esc(s.vo.who)+'</span>“'+esc(s.vo.line)+'”'; }
    else { vo.innerHTML = '<span class="an-vo-who">SFX</span><i>music &amp; sound only — no dialogue</i>'; }
    vo.classList.add('show');
    score.textContent = '♪ temp score — ' + s.motif;
    count.textContent = 'Shot ' + (cur+1) + ' / ' + SHOTS.length;
    shotT = 0; fill.style.width = '0%';
    Array.prototype.forEach.call(strip.children, function(c,idx){ c.classList.toggle('active', idx===cur); });
    renderCmts();
    if (playMotif) motif(s.motif);
  }

  function frameStep(now){
    if (!playing) return;
    var dt = Math.min(0.05, (now-lastTs)/1000); lastTs = now;
    shotT += dt;
    var s = SHOTS[cur];
    fill.style.width = Math.min(100, shotT/s.dur*100) + '%';
    if (shotT >= s.dur){
      if (cur === SHOTS.length-1){ pause(); fill.style.width='100%'; return; }
      showShot(cur+1, true);
    }
    raf = requestAnimationFrame(frameStep);
  }
  function play(){ if(playing) return; playing=true; playBtn.textContent='⏸'; playBtn.classList.add('on'); playBtn.setAttribute('aria-label','Pause reel'); lastTs=performance.now(); motif(SHOTS[cur].motif); raf=requestAnimationFrame(frameStep); }
  function pause(){ playing=false; playBtn.textContent='▶'; playBtn.classList.remove('on'); playBtn.setAttribute('aria-label','Play reel'); if(raf) cancelAnimationFrame(raf); }

  playBtn.onclick = function(){ playing ? pause() : play(); };
  $('an-prev').onclick = function(){ showShot(cur-1, true); if(playing) lastTs=performance.now(); };
  $('an-next').onclick = function(){ showShot(cur+1, true); if(playing) lastTs=performance.now(); };
  muteBtn.onclick = function(){ muted=!muted; muteBtn.classList.toggle('muted', muted); muteBtn.title = muted?'Temp score muted':'Temp score on'; };

  // filmstrip
  SHOTS.forEach(function(s, idx){

    b.innerHTML = '<span class="an-thumb-art">'+s.art+'</span><span class="an-thumb-n">'+(idx+1)+'</span>';
    b.onclick = function(){ showShot(idx, true); if(playing) lastTs=performance.now(); };
    strip.appendChild(b);
  });

  // ---- per-shot comments (local) ----
  function cmtKey(id){ return 'pjcc.animatic.cmt.' + id; }
  function getCmts(id){ try { return JSON.parse(localStorage.getItem(cmtKey(id)) || '[]'); } catch(e){ return []; } }
  function saveCmts(id, arr){ try { localStorage.setItem(cmtKey(id), JSON.stringify(arr)); } catch(e){} }
  function cmtEl(name, text, pin){
    var d = document.createElement('div'); d.className = 'an-cmt' + (pin ? ' pin' : '');
    d.innerHTML = '<span class="an-cmt-author">' + esc(name||'anon') + (pin ? ' <em>· director</em>' : '') + '</span>' + esc(text);
    return d;
  }
  function renderCmts(){
    var s = SHOTS[cur];
    $('an-cmts-shot').textContent = s.slug;
    var list = $('an-cmts-list'); list.innerHTML = '';
    var pinned = PINNED[s.id] || [];
    pinned.forEach(function(c){ list.appendChild(cmtEl(c.name, c.text, true)); });
    var arr = getCmts(s.id);
    if (!pinned.length && !arr.length){ list.innerHTML = '<div class="an-cmt-empty">No notes on this frame yet. Be the first to weigh in.</div>'; }
    arr.slice().reverse().forEach(function(c){ list.appendChild(cmtEl(c.name, c.text, false)); });
  }
  function postCmt(){
    var t = $('an-cmt-text').value.trim(); if (!t) return;
    var n = $('an-cmt-name').value.trim() || 'anon';
    var s = SHOTS[cur]; var arr = getCmts(s.id);
    arr.push({ name:n, text:t, ts:Date.now() }); saveCmts(s.id, arr);
    $('an-cmt-text').value = ''; renderCmts();
  }
  $('an-cmt-send').onclick = postCmt;
  $('an-cmt-text').addEventListener('keydown', function(e){ if (e.key === 'Enter'){ e.preventDefault(); postCmt(); } });

  showShot(0, false);
})();
</script>
